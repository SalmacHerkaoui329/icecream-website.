create database if not exists clients;
use clients;
create table client(
codeclient int primary key,
nom varchar(40),
prenom varchar(40),
adresse varchar(50),
ville varchar(40),
telephone int 
);
create table commande (
numc int primary key ,
datee int ,
codeclient int,
Foreign key (codeclient) references client(codeclient)
);
create table article (
codeA int primary key,
nom varchar(45) ,
QTS int ,
prix float ,
coleur varchar(45),
prixvent float 
);
create table ligne_commande(
numc int ,
codeA int ,
QteComm int,
prix_total_comande float,
Foreign key (numc) references commande(numc),
Foreign key (codeA) references article(codeA)
);
create table fournisseur (
cofeF int  primary key ,
nom varchar (45),
adresse varchar (45),
ville varchar (45)
);
create table entreprise_distrubution (
nom_entreprise varchar(45) primary key,
telephone int ,
fax int ,
cofeF int,
Foreign key (cofeF) references fournisseur(cofeF)
);
create table livraison (
num int  primary key ,
datee int ,
codeA int ,
Qtelivr int ,
nom_entreprise varchar(45),
Foreign key (codeA) references article(codeA),
Foreign key (nom_entreprise) references  entreprise_distrubution(nom_entreprise)
);
